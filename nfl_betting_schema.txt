# NFL Betting Analytics Platform - Complete Build Guide

## Project Overview
A full-stack NFL betting analytics platform that surfaces betting odds alongside team statistics, advanced metrics, and situational data to help users make informed betting decisions. The platform tracks user bets to help identify personal biases and tendencies.

**Tech Stack:**
- Frontend: Next.js 14+ (App Router), React, TypeScript, Tailwind CSS, shadcn/ui
- Backend: Next.js API routes
- Database: PostgreSQL (via Supabase free tier)
- Data Sources: The Odds API, ESPN API, Pro Football Reference
- Deployment: Vercel (free tier)

---

## Database Schema

### Core Tables

```sql
-- Teams table
CREATE TABLE teams (
  id SERIAL PRIMARY KEY,
  external_id VARCHAR(50) UNIQUE NOT NULL, -- ESPN team ID
  name VARCHAR(100) NOT NULL,
  abbreviation VARCHAR(10) NOT NULL,
  conference VARCHAR(10),
  division VARCHAR(20),
  logo_url TEXT,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Games table
CREATE TABLE games (
  id SERIAL PRIMARY KEY,
  external_id VARCHAR(50) UNIQUE NOT NULL, -- ESPN game ID
  season INTEGER NOT NULL,
  week INTEGER NOT NULL,
  game_date TIMESTAMP NOT NULL,
  home_team_id INTEGER REFERENCES teams(id),
  away_team_id INTEGER REFERENCES teams(id),
  home_score INTEGER,
  away_score INTEGER,
  status VARCHAR(20), -- 'scheduled', 'in_progress', 'final'
  venue VARCHAR(200),
  weather_temp DECIMAL(5,2),
  weather_condition VARCHAR(50),
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  CONSTRAINT unique_game UNIQUE(external_id)
);

-- Odds table (snapshot approach for line movement tracking)
CREATE TABLE odds (
  id SERIAL PRIMARY KEY,
  game_id INTEGER REFERENCES games(id),
  bookmaker VARCHAR(50) NOT NULL,
  market_type VARCHAR(20) NOT NULL, -- 'spreads', 'totals', 'moneyline'
  home_odds DECIMAL(10,2),
  away_odds DECIMAL(10,2),
  home_point DECIMAL(5,1), -- spread or total
  away_point DECIMAL(5,1),
  timestamp TIMESTAMP DEFAULT NOW(),
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_odds_game ON odds(game_id);
CREATE INDEX idx_odds_timestamp ON odds(timestamp);

-- Team season stats (aggregate stats updated weekly)
CREATE TABLE team_stats (
  id SERIAL PRIMARY KEY,
  team_id INTEGER REFERENCES teams(id),
  season INTEGER NOT NULL,
  week INTEGER NOT NULL,
  
  -- Basic offense
  points_per_game DECIMAL(5,2),
  total_yards_per_game DECIMAL(6,2),
  pass_yards_per_game DECIMAL(6,2),
  rush_yards_per_game DECIMAL(6,2),
  
  -- Basic defense
  points_allowed_per_game DECIMAL(5,2),
  total_yards_allowed_per_game DECIMAL(6,2),
  pass_yards_allowed_per_game DECIMAL(6,2),
  rush_yards_allowed_per_game DECIMAL(6,2),
  
  -- Efficiency
  third_down_pct DECIMAL(5,2),
  third_down_defense_pct DECIMAL(5,2),
  red_zone_pct DECIMAL(5,2),
  red_zone_defense_pct DECIMAL(5,2),
  
  -- Turnovers
  turnovers_gained INTEGER,
  turnovers_lost INTEGER,
  turnover_differential INTEGER,
  
  -- Advanced (if available)
  epa_per_play_offense DECIMAL(6,3),
  epa_per_play_defense DECIMAL(6,3),
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  CONSTRAINT unique_team_week UNIQUE(team_id, season, week)
);

CREATE INDEX idx_team_stats_lookup ON team_stats(team_id, season, week);

-- Home/Away splits
CREATE TABLE team_splits (
  id SERIAL PRIMARY KEY,
  team_id INTEGER REFERENCES teams(id),
  season INTEGER NOT NULL,
  location VARCHAR(10) NOT NULL, -- 'home' or 'away'
  
  games_played INTEGER DEFAULT 0,
  wins INTEGER DEFAULT 0,
  losses INTEGER DEFAULT 0,
  points_per_game DECIMAL(5,2),
  points_allowed_per_game DECIMAL(5,2),
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  CONSTRAINT unique_team_split UNIQUE(team_id, season, location)
);

-- User bets tracking
CREATE TABLE user_bets (
  id SERIAL PRIMARY KEY,
  user_id UUID, -- For future auth, nullable for now
  game_id INTEGER REFERENCES games(id),
  bet_type VARCHAR(20) NOT NULL, -- 'spread', 'moneyline', 'total_over', 'total_under'
  team_id INTEGER REFERENCES teams(id), -- NULL for totals
  odds DECIMAL(10,2) NOT NULL,
  stake DECIMAL(10,2) NOT NULL,
  potential_win DECIMAL(10,2),
  result VARCHAR(20), -- 'pending', 'won', 'lost', 'push'
  notes TEXT,
  placed_at TIMESTAMP DEFAULT NOW(),
  settled_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_user_bets_user ON user_bets(user_id);
CREATE INDEX idx_user_bets_game ON user_bets(game_id);

-- User bet analytics (computed/cached)
CREATE TABLE user_bet_analytics (
  id SERIAL PRIMARY KEY,
  user_id UUID UNIQUE,
  
  total_bets INTEGER DEFAULT 0,
  total_wins INTEGER DEFAULT 0,
  total_losses INTEGER DEFAULT 0,
  total_pushes INTEGER DEFAULT 0,
  win_rate DECIMAL(5,2),
  
  total_staked DECIMAL(10,2) DEFAULT 0,
  total_returned DECIMAL(10,2) DEFAULT 0,
  net_profit DECIMAL(10,2) DEFAULT 0,
  roi DECIMAL(5,2),
  
  -- Bias indicators
  favorite_team_id INTEGER REFERENCES teams(id),
  favorite_team_bet_count INTEGER DEFAULT 0,
  favorite_team_win_rate DECIMAL(5,2),
  
  avg_stake DECIMAL(10,2),
  biggest_win DECIMAL(10,2),
  biggest_loss DECIMAL(10,2),
  
  updated_at TIMESTAMP DEFAULT NOW()
);
```

## API Endpoints Structure

### Data Ingestion (Backend Cron Jobs)
```
POST /api/cron/fetch-odds         # Fetch current odds from The Odds API
POST /api/cron/fetch-games         # Fetch schedule/scores from ESPN
POST /api/cron/fetch-team-stats    # Fetch/calculate team statistics
POST /api/cron/update-game-status  # Update live scores
```

### Public API Endpoints
```
GET /api/games                     # List games (with filters)
  ?week=1&season=2024&team=PHI
  
GET /api/games/[gameId]           # Game details with odds & stats
  Returns: game info, both teams' stats, current odds, line movement

GET /api/teams                     # List all teams

GET /api/teams/[teamId]           # Team details
  Returns: team info, current season stats, home/away splits

GET /api/teams/[teamId]/stats     # Historical stats by week
  ?season=2024

GET /api/odds/[gameId]            # Odds history for a game
  Returns: line movement over time

GET /api/matchup                   # Compare two teams
  ?homeTeam=PHI&awayTeam=DAL&week=5
  Returns: head-to-head stats, matchup analysis
```

### User Endpoints (Phase 3)
```
POST /api/bets                     # Track a new bet
GET /api/bets                      # Get user's bets
GET /api/bets/analytics            # Get user analytics/biases
PATCH /api/bets/[betId]           # Update bet result
```
